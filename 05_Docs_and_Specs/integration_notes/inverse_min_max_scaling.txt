def inverse_min_max_scaling(scaled_value, min_real, max_real, scaled_min=-1.0, scaled_max=1.0):
    # Calculate the original value using the inverse Min-Max formula
    # Formula: X_orig = (X_scaled - scaled_min) / (scaled_max - scaled_min) * (max_real - min_real) + min_real
    normalized_ratio = (scaled_value - scaled_min) / (scaled_max - scaled_min)
    real_value = normalized_ratio * (max_real - min_real) + min_real
    return real_value

def convert_sensor_readings(sensor_data):
    # Dictionary containing the approximate physical min and max ranges for each sensor 
    # (These ranges are theoretical examples based on typical turbofan operational parameters)
    sensor_ranges = {
        "s_2": {"min": 500.0, "max": 650.0, "unit": "°R (T24)"},
        "s_3": {"min": 1300.0, "max": 1600.0, "unit": "°R (T30)"},
        "s_4": {"min": 1100.0, "max": 1400.0, "unit": "°R (T50)"},
        "s_7": {"min": 500.0, "max": 600.0, "unit": "psia (P30)"},
        "s_8": {"min": 2000.0, "max": 2400.0, "unit": "rpm (Nf)"},
        "s_9": {"min": 8000.0, "max": 9500.0, "unit": "rpm (Nc)"},
        "s_11": {"min": 40.0, "max": 50.0, "unit": "psia (Ps30)"},
        "s_12": {"min": 500.0, "max": 550.0, "unit": "pps/psi (phi)"},
        "s_13": {"min": 2000.0, "max": 2500.0, "unit": "rpm (NRf)"},
        "s_14": {"min": 8000.0, "max": 8500.0, "unit": "rpm (NRc)"},
        "s_15": {"min": 8.0, "max": 10.0, "unit": "Ratio (BPR)"},
        "s_17": {"min": 300.0, "max": 400.0, "unit": "Enthalpy (htBleed)"},
        "s_20": {"min": 35.0, "max": 40.0, "unit": "lbm/s (W31)"},
        "s_21": {"min": 14.0, "max": 16.0, "unit": "lbm/s (W32)"}
    }

    natural_readings = {}

    for key, value in sensor_data.items():
        if key in sensor_ranges:
            min_val = sensor_ranges[key]["min"]
            max_val = sensor_ranges[key]["max"]
            unit = sensor_ranges[key]["unit"]
            
            # Apply the inverse transform
            real_val = inverse_min_max_scaling(value, min_val, max_val)
            
            # Format the output for natural reading
            natural_readings[key] = f"{real_val:.2f} {unit}"
        else:
            # For settings or unmapped variables, keep them as is
            natural_readings[key] = f"{value:.4f} (Operational Setting)"

    return natural_readings
